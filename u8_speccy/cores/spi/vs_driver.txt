;------------------------------------------------------------------------------
; VS1053 Init
;------------------------------------------------------------------------------
VS_INIT:        LD  A, %00000000          ; XCS=0 XDCS=0
                OUT (#05), A
                LD  HL, TABLE
                LD  B, 44
VS_INIT1:       LD  E, (HL)
                CALL VS_SDI
                JR  C, ERROR
                INC HL
                DJNZ VS_INIT1
                LD  A, %00100000          ; XCS=0 XDCS=1
                OUT (#05), A
ERROR:          RET

;------------------------------------------------------------------------------
; VS1053
;------------------------------------------------------------------------------
; #xx04 Data Buffer (write/read):
;     bit 7-0 = Stores read/write data
;
; #xx05 Command/Status Register (write):
;     bit 7   = XSC
;     bit 6   = XDCS
;     bit 5   = mode      0: SCI/SDI mode; 1: STREAM 32bit (data word + data word)
;     bit 4   = spiclk    0: 1.75MHz; 1: 3.5MHz
;     bit 3-0 = Reserved
;
; #xx05 Command/Status Register (read):
;     bit 7   = BUSY      1: busy, transfer in progress; 0: idle
;     bit 6   = DREQ      1: data not requested; 0: data requested
;     bit 5-0 = Reserved
;------------------------------------------------------------------------------

;------------------------------------------------------------------------------
; VS_PUT - Send data stream to VS1053
; HL = data pointer, BC = byte count
;------------------------------------------------------------------------------
VS_PUT:         LD  A, (HL)
                CALL VS_WR
                JR  NC, vs_error          ; CY=0 ERROR!
                INC HL
                DEC BC
                LD  A, C
                OR  B
                JR  NZ, VS_PUT
vs_error:       RET

;------------------------------------------------------------------------------
; VS_SCI - Send SCI command (4 bytes)
; HL = address of command
;------------------------------------------------------------------------------
VS_SCI:         LD  B, 4
vs_sci1:        LD  A, (HL)
                CALL VS_WR
                JR  NC, vs_error
                INC HL
                DJNZ vs_sci1
                RET

;------------------------------------------------------------------------------
; VS_WR - Write single byte to VS1053 with busy wait
; A = data 7..0
; Returns: CY=1 OK, CY=0 Timeout
;------------------------------------------------------------------------------
VS_WR:          OUT (#04), A              ; data out 7..0
                LD  DE, #0000
vs_wait:        DEC DE
                LD  A, E
                OR  D
                RET Z                     ; CY=0 Time Out
                IN  A, (#05)
                RLCA
                JR  C, vs_wait            ; Wait while BUSY
                RLCA
                JR  NC, vs_wait           ; Wait while DREQ not ready
                IN  A, (#04)              ; CY=1 OK
                RET

;------------------------------------------------------------------------------
; Data Tables
;------------------------------------------------------------------------------
DATA_RESET:     DB  #02, #00, %01001000, %00000100  ; [WRITE][SCI_MODE][SM_SDINEW & SM_LINE1][SM_RESET]

                DB  #80, #BB, #00, #00              ; 48kHz sample rate
                DB  #00, #EE, #02, #00

                DB  #04, #00
